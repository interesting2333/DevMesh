import test from 'node:test';
import assert from 'node:assert/strict';
import {createHash} from 'node:crypto';
import {mkdir,writeFile,readFile,rm} from 'node:fs/promises';
import {dirname,join} from 'node:path';
import {LocalContentStore} from '../packages/blob-store/dist/index.mjs';
import {ControlledGit,LocalSourceArchives,LocalWorkspacePreservations,WorkspaceError} from '../packages/code-workspaces/dist/index.mjs';
import {archiveGit,archiveOp,sourceArchiveFixture} from './fixtures/source-archive.mjs';
const hash=value=>createHash('sha256').update(value).digest('hex');

test('actual isolated Git source and complete workspace bytes require original preservation proof before reclamation',async t=>{
 const f=await sourceArchiveFixture(t),content=new LocalContentStore(f.storage,f.local.dataRoot,0),sources=new LocalSourceArchives(f.options),service=new LocalWorkspacePreservations({...f.options,content,sourceArchives:sources}),parent=dirname(f.binding.root),input={permit:f.permit,operationId:archiveOp(),allocationOperationId:f.allocation.record.operationId};
 await writeFile(join(f.binding.root,'draft.txt'),'unsubmitted pre-commit bytes');await mkdir(join(parent,'tmp'));await writeFile(join(parent,'tmp','compiler.log'),'recovery output');const projectRoot=f.repository.provider.identity.root;let saved;
 let beforeCommit;await t.test('a workspace with no repo.commit retains the admitted base and all unsubmitted bytes',async()=>{
  beforeCommit=await service.archive({...input,operationId:archiveOp()});assert.equal(beforeCommit.state,'ACKNOWLEDGED');assert.equal(beforeCommit.basis.commits.length,0);assert.equal(beforeCommit.manifest.retained.length,1);assert.equal(beforeCommit.manifest.retained[0].commit,f.base);assert.equal(await archiveGit(projectRoot,'rev-parse',beforeCommit.manifest.retained[0].ref),f.base);assert.equal(f.sql('SELECT count(*) n FROM source_archives')[0].n,0);assert.ok(beforeCommit.manifest.snapshot.entries.some(e=>e.kind==='FILE'&&e.path==='repo/draft.txt'));const held=await service.acquireReclaimable({permit:f.permit,operationId:beforeCommit.operationId,manifestSha256:beforeCommit.acknowledgement.manifestSha256});await held.release();
 });
 const committed=await f.commit();await assert.rejects(service.acquireReclaimable({permit:f.permit,operationId:beforeCommit.operationId,manifestSha256:beforeCommit.acknowledgement.manifestSha256}),{code:'WORKSPACE_PRESERVATION_BASIS_CHANGED'});await writeFile(join(f.binding.root,'draft.txt'),'uncommitted original bytes');
 await t.test('interrupted publication retains its original operation, every source commit and immutable material bytes',async()=>{
  const publish=content.provider.publish.bind(content.provider);let interrupted=true;content.provider.publish=async write=>{if(interrupted&&write.expectedHash===hash('uncommitted original bytes')){interrupted=false;throw Error('Material publish interrupted');}return publish(write);};
  await assert.rejects(service.archive(input),/Material publish interrupted/);const pending=await f.storage.request('getWorkspacePreservation',{permit:f.permit,operationId:input.operationId});assert.equal(pending.state,'PREPARED');assert.equal(pending.basis.commits.length,1);assert.equal(pending.basis.commits[0].operationId,committed.operationId);assert.equal(pending.basis.commits[0].archive.acknowledgement.commit,committed.result.commit);
  await assert.rejects(service.acquireReclaimable({permit:f.permit,operationId:input.operationId,manifestSha256:hash('not an ACK')}),{code:'WORKSPACE_PRESERVATION_ACK_REQUIRED'});assert.equal(await readFile(join(f.binding.root,'draft.txt'),'utf8'),'uncommitted original bytes');const originalWrite=f.sql("SELECT write_id FROM object_write_operations WHERE purpose='workspacePreservation' AND expected_hash=?",hash('uncommitted original bytes'))[0].write_id;
  const native=ControlledGit.prototype.updateRefs;let writes=0;ControlledGit.prototype.updateRefs=async()=>{writes++;throw Error('Recovery must not mutate original Git refs');};try{saved=await service.reconcile({permit:f.permit,operationId:input.operationId});}finally{ControlledGit.prototype.updateRefs=native;}assert.equal(writes,0);assert.equal(saved.state,'ACKNOWLEDGED');assert.equal(saved.operationId,input.operationId);assert.equal(f.sql("SELECT write_id FROM object_write_operations WHERE purpose='workspacePreservation' AND expected_hash=?",hash('uncommitted original bytes'))[0].write_id,originalWrite);
  const rows=await f.storage.request('listWorkspacePreservationMaterials',{permit:f.permit,operationId:input.operationId,afterKey:'',limit:100}),draft=rows.find(row=>row.sha256===hash('uncommitted original bytes'));assert.equal(Buffer.from(await content.readBytes('user_1',{kind:'workspacePreservation',operationId:input.operationId,materialKey:draft.materialKey})).toString(),'uncommitted original bytes');assert.ok(saved.manifest.snapshot.entries.some(e=>e.path==='tmp/compiler.log'&&e.kind==='FILE'));
  for(const held of saved.manifest.retained)assert.equal(await archiveGit(projectRoot,'rev-parse',held.ref),held.commit);assert.equal(await f.allocation.provider.head(),f.base);assert.equal(f.sql('SELECT accepted_release_id FROM projects')[0].accepted_release_id,null);assert.equal(f.held.size,0);
 });
 await t.test('unknown refs, staged bytes, missing content and a busy Runner boundary independently block reclamation',async()=>{
  const request={permit:f.permit,operationId:input.operationId,manifestSha256:saved.acknowledgement.manifestSha256},lease=await service.acquireReclaimable(request);assert.equal(lease.parent,parent);await lease.release();
  await archiveGit(f.binding.root,'update-ref','refs/heads/unretained-fixture',f.base);await assert.rejects(service.acquireReclaimable(request),{code:'GIT_PRESERVATION_UNRETAINED_REF'});await archiveGit(f.binding.root,'update-ref','-d','refs/heads/unretained-fixture');
  await archiveGit(f.binding.root,'add','src/board.ts');await assert.rejects(service.acquireReclaimable(request),{code:'GIT_PRESERVATION_INDEX_CHANGED'});await archiveGit(f.binding.root,'reset','--mixed',f.base);
  const digest=hash('uncommitted original bytes'),path=join(f.local.dataRoot,'artifacts','sha256',digest.slice(0,2),digest),bytes=await readFile(path);await rm(path);await assert.rejects(service.acquireReclaimable(request),{code:'CONTENT_MISSING'});await writeFile(path,bytes);
  const acquire=f.options.fencer.acquire.bind(f.options.fencer);f.options.fencer.acquire=async(...args)=>{if(args[0].kind==='workspace')throw new WorkspaceError('RUNNER_WORKSPACE_BUSY');return acquire(...args);};try{await assert.rejects(service.acquireReclaimable(request),{code:'RUNNER_WORKSPACE_BUSY'});}finally{f.options.fencer.acquire=acquire;}assert.equal(f.held.size,0);assert.equal(await readFile(join(parent,'tmp','compiler.log'),'utf8'),'recovery output');
 });
 await t.test('hash drift stays preserved and a lost complete ACK cannot turn unknown recovery material into deletion permission',async()=>{
  await writeFile(join(f.binding.root,'draft.txt'),'later changed bytes');await assert.rejects(service.acquireReclaimable({permit:f.permit,operationId:input.operationId,manifestSha256:saved.acknowledgement.manifestSha256}),{code:'WORKSPACE_MATERIAL_CHANGED'});
  await mkdir(join(parent,'.devmesh-file-operations','unknown-owner'),{recursive:true});await writeFile(join(parent,'.devmesh-file-operations','unknown-owner','previous'),'unknown original recovery');const next={...input,operationId:archiveOp()},native=f.storage.request.bind(f.storage);let lost=true;f.storage.request=async(method,args)=>{const result=await native(method,args);if(method==='completeWorkspacePreservation'&&lost){lost=false;throw Error('Complete preservation ACK reply lost');}return result;};let recovered;try{recovered=await service.archive(next);}finally{f.storage.request=native;}
  assert.equal(lost,false);assert.equal(recovered.state,'ACKNOWLEDGED');assert.equal(recovered.operationId,next.operationId);assert.equal(f.sql('SELECT count(*) n FROM workspace_preservations')[0].n,3);assert.equal(f.sql('SELECT count(*) n FROM source_archives')[0].n,1);assert.ok(recovered.manifest.snapshot.entries.some(e=>e.kind==='FILE'&&e.path==='.devmesh-file-operations/unknown-owner/previous'));
  await assert.rejects(service.acquireReclaimable({permit:f.permit,operationId:next.operationId,manifestSha256:recovered.acknowledgement.manifestSha256}),{code:'WORKSPACE_PRESERVATION_RECOVERY_UNKNOWN'});assert.equal(await readFile(join(parent,'.devmesh-file-operations','unknown-owner','previous'),'utf8'),'unknown original recovery');assert.equal(await readFile(join(f.binding.root,'draft.txt'),'utf8'),'later changed bytes');assert.equal(f.sql('SELECT accepted_release_id FROM projects')[0].accepted_release_id,null);assert.deepEqual(f.sql('PRAGMA foreign_key_check'),[]);assert.equal(f.held.size,0);
 });
});
