import {randomUUID} from 'node:crypto';
import type {DatabaseSync,SQLInputValue} from 'node:sqlite';
import {parseContract} from '@devmesh/contracts';
import type {ExecutionPermit} from '@devmesh/contracts';
import {canonical} from './json.js';
import {digest,transaction} from './migrations.js';
import {StorageError} from './errors.js';
import {createWorkspaceDomain} from './workspace-domain.js';
import {createToolAuthorityDomain} from './tool-authority-domain.js';
import {workspacePreservationRequests} from './workspace-preservation-protocol.js';
import type {WorkspacePreservation,WorkspacePreservationInput,WorkspacePreservationMethod,WorkspacePreservationInputs,WorkspacePreservationMaterial,WorkspacePreservationManifest} from './workspace-preservation-protocol.js';
import type {ManagedRepository,WorkspaceAllocation} from './workspace-protocol.js';
type Row=Record<string,string|number|null>;
type Target={operationId:string;materialKey:string};
function fail(code:string,status=409):never{throw new StorageError(code,status,'Complete original workspace preservation is required');}
const same=(a:unknown,b:unknown)=>canonical(a)===canonical(b);
export function workspacePreservationMaterialKey(path:string){return digest(path);}
export function createWorkspacePreservationDomain(db:DatabaseSync){
 const get=(sql:string,...p:SQLInputValue[])=>db.prepare(sql).get(...p) as Row|undefined,all=(sql:string,...p:SQLInputValue[])=>db.prepare(sql).all(...p) as Row[],exec=(sql:string,...p:SQLInputValue[])=>db.prepare(sql).run(...p);
 const workspaces=createWorkspaceDomain(db),authority=createToolAuthorityDomain(db);
 const required=(id:string)=>get('SELECT * FROM workspace_preservations WHERE operation_id=?',id)??fail('WORKSPACE_PRESERVATION_NOT_FOUND',404);
 function membership(projectId:string,principalId:string,write:boolean){const m=get('SELECT role FROM project_members WHERE project_id=? AND principal_id=?',projectId,principalId);if(!m||write&&m.role==='viewer')fail('WORKSPACE_PRESERVATION_NOT_FOUND',404);}
 function owner(r:Row,p:ExecutionPermit){const h=authority.check(p,'repo.by-operation','read');membership(String(r.project_id),h.principalId,true);if(h.principalId!==r.principal_id)fail('WORKSPACE_PRESERVATION_NOT_FOUND',404);const a=workspaces.execute('getWorkspaceAllocation',{permit:p,operationId:String(r.allocation_operation_id)}) as WorkspaceAllocation;if(a.workspaceRef!==r.workspace_ref)fail('WORKSPACE_PRESERVATION_NOT_FOUND',404);return r;}
 const view=(r:Row):WorkspacePreservation=>({operationId:String(r.operation_id),allocationOperationId:String(r.allocation_operation_id),workspaceRef:String(r.workspace_ref),principalId:String(r.principal_id),requestHash:String(r.request_hash),basisSha256:String(r.basis_hash),state:r.state as WorkspacePreservation['state'],basis:JSON.parse(String(r.basis_json)),manifest:r.manifest_json?JSON.parse(String(r.manifest_json)):null,acknowledgement:r.acknowledgement_json?JSON.parse(String(r.acknowledgement_json)):null,acknowledgedAt:r.acknowledged_at as string|null});
 function inputs(i:{permit:ExecutionPermit;allocationOperationId:string}):WorkspacePreservationInputs{
  const h=authority.check(i.permit,'repo.by-operation','read'),allocation=workspaces.execute('getWorkspaceAllocation',{permit:i.permit,operationId:i.allocationOperationId}) as WorkspaceAllocation,repository=workspaces.execute('getAllocationRepository',{permit:i.permit,operationId:i.allocationOperationId}) as ManagedRepository;
  membership(allocation.projectId,h.principalId,true);if(allocation.projectId!==h.projectId||allocation.state!=='READY'||!allocation.identity||!allocation.manifest||repository.state!=='READY'||!repository.identity)fail('WORKSPACE_PRESERVATION_SOURCE_UNAVAILABLE');
  const p=i.permit;if(p.ownerKind!=='workflowAttempt'||!p.resources.some(r=>r.kind==='workspace'&&r.resourceId===allocation.workspaceRef)||!p.resources.some(r=>r.kind==='projectRepository'&&r.resourceId===repository.resourceId))fail('WORKSPACE_PRESERVATION_OWNER_DENIED',403);
  const operations=all('SELECT * FROM repo_operations WHERE attempt_id=? ORDER BY operation_id',allocation.attemptId);
  if(operations.some(r=>r.run_id!==allocation.runId||r.activation_id!==allocation.activationId||r.grant_id!==allocation.grantId||!['SUCCEEDED','FAILED'].includes(String(r.state))))fail('WORKSPACE_PRESERVATION_GIT_UNKNOWN');
  const commits:WorkspacePreservationInputs['commits']=[];for(const r of operations){if(r.tool_name!=='repo.commit'||r.state!=='SUCCEEDED')continue;const prepared=r.prepared_json?JSON.parse(String(r.prepared_json)):null,result=r.result_json?JSON.parse(String(r.result_json)):null,request=JSON.parse(String(r.request_json));if(!prepared||result?.commit!==prepared.commit||prepared.ref!=='refs/devmesh/operations/'+r.operation_id||request.binding.workspaceRef!==allocation.workspaceRef||request.binding.baseCommit!==allocation.baseCommit)fail('WORKSPACE_PRESERVATION_SOURCE_UNPROVEN');
   const saved=all("SELECT * FROM source_archives WHERE allocation_operation_id=? AND commit_operation_id=? AND state='ACKNOWLEDGED' ORDER BY rowid",allocation.operationId,r.operation_id!).at(0);commits.push({operationId:String(r.operation_id),requestHash:String(r.request_hash),commit:prepared.commit,tree:prepared.tree,ref:prepared.ref,archive:saved?{operationId:String(saved.operation_id),manifest:JSON.parse(String(saved.manifest_json)),acknowledgement:JSON.parse(String(saved.acknowledgement_json))}:null});
  }
  if(commits.length>64)fail('WORKSPACE_PRESERVATION_TOO_MANY_COMMITS');
  const files=all('SELECT * FROM repo_file_operations WHERE workspace_ref=? ORDER BY operation_id',allocation.workspaceRef);if(files.some(r=>r.run_id!==allocation.runId||r.attempt_id!==allocation.attemptId||r.activation_id!==allocation.activationId||r.grant_id!==allocation.grantId)||files.length>8000)fail('WORKSPACE_PRESERVATION_OWNER_DENIED',403);
  return {principalId:h.principalId,allocation,repository,commits,fileOperations:files.map(r=>({operationId:String(r.operation_id),requestHash:String(r.request_hash),state:String(r.state),cleanupConfirmed:r.cleanup_confirmed===1,proof:r.prepared_json?JSON.parse(String(r.prepared_json)):null,result:r.result_json?JSON.parse(String(r.result_json)):null}))};
 }
 function current(r:Row,permit:ExecutionPermit){owner(r,permit);if(digest(canonical(inputs({permit,allocationOperationId:String(r.allocation_operation_id)})))!==r.basis_hash)fail('WORKSPACE_PRESERVATION_BASIS_CHANGED');}
 const material=(t:Target)=>get('SELECT * FROM workspace_preservation_materials WHERE operation_id=? AND material_key=?',t.operationId,t.materialKey)??fail('WORKSPACE_PRESERVATION_MATERIAL_NOT_FOUND',404);
 const materialView=(r:Row):WorkspacePreservationMaterial=>({operationId:String(r.operation_id),materialKey:String(r.material_key),sha256:String(r.expected_hash),byteSize:Number(r.byte_size),writeKey:String(r.write_key),contentHash:r.content_hash as string|null,writeId:r.write_id as string|null});
 function contentReady(operationId:string){const items=all('SELECT * FROM workspace_preservation_materials WHERE operation_id=?',operationId);if(!items.length||items.some(r=>!r.content_hash||!get("SELECT 1 FROM content_objects WHERE content_hash=? AND byte_size=? AND state='READY'",r.expected_hash!,r.byte_size!)))fail('WORKSPACE_PRESERVATION_MATERIAL_UNAVAILABLE');return items.length;}
 function validateManifest(r:Row,m:WorkspacePreservationManifest){const b=view(r).basis,a=b.allocation,repo=b.repository,{sha256,...snapshot}=m.snapshot;
  if(m.operationId!==r.operation_id||m.allocationOperationId!==a.operationId||m.workspaceRef!==a.workspaceRef||m.basisSha256!==r.basis_hash||m.headCommit!==a.baseCommit||!same(m.repositoryIdentity,repo.identity)||!same(m.workspaceIdentity,a.identity)||digest(canonical(snapshot))!==sha256||m.snapshot.byteSize!==m.snapshot.entries.reduce((n,e)=>n+(e.kind==='FILE'?e.byteSize:0),0)||new Set(m.snapshot.entries.map(e=>e.path.toLowerCase())).size!==m.snapshot.entries.length||Buffer.byteLength(canonical(m))>8*1024*1024)fail('WORKSPACE_PRESERVATION_MANIFEST_CHANGED');
  const expected=[{ref:'refs/devmesh/workspace-preservations/'+r.operation_id+'/base',commit:a.baseCommit,manifest:a.manifest},...b.commits.map(c=>{if(!c.archive)fail('WORKSPACE_PRESERVATION_SOURCE_UNRETAINED');return {ref:c.archive.acknowledgement.archiveRef,commit:c.commit,manifest:c.archive.manifest};})];if(!same(m.retained,expected))fail('WORKSPACE_PRESERVATION_SOURCE_UNRETAINED');
 }
 const handlers={
  getWorkspacePreservationInputs(i:WorkspacePreservationInput<'getWorkspacePreservationInputs'>){return inputs(i);},
  beginWorkspacePreservation(i:WorkspacePreservationInput<'beginWorkspacePreservation'>){return transaction(db,()=>{const old=get('SELECT * FROM workspace_preservations WHERE operation_id=?',i.operationId);if(old){owner(old,i.permit);if(old.allocation_operation_id!==i.allocationOperationId)fail('IDEMPOTENCY_KEY_REUSED');return {...view(old),created:false};}
   if(['repo_operations','repo_file_operations','workspace_allocations','source_archives','candidate_integrations'].some(table=>get('SELECT 1 FROM '+table+' WHERE operation_id=?',i.operationId)))fail('IDEMPOTENCY_KEY_REUSED');const b=inputs(i);if(b.commits.some(c=>!c.archive))fail('WORKSPACE_PRESERVATION_SOURCE_UNRETAINED');const basis=canonical(b),basisHash=digest(basis),requestHash=digest(canonical({allocationOperationId:i.allocationOperationId,principalId:b.principalId}));if(get("SELECT 1 FROM workspace_preservations WHERE workspace_ref=? AND state<>'ACKNOWLEDGED'",b.allocation.workspaceRef))fail('WORKSPACE_PRESERVATION_RECONCILIATION_REQUIRED');
   exec("INSERT INTO workspace_preservations(operation_id,allocation_operation_id,workspace_ref,project_id,principal_id,request_hash,basis_hash,basis_json,state,created_at) VALUES(?,?,?,?,?,?,?,?,'SENT',?)",i.operationId,i.allocationOperationId,b.allocation.workspaceRef,b.allocation.projectId,b.principalId,requestHash,basisHash,basis,new Date().toISOString());return {...view(required(i.operationId)),created:true};});},
  getWorkspacePreservation(i:WorkspacePreservationInput<'getWorkspacePreservation'>){return view(owner(required(i.operationId),i.permit));},
  prepareWorkspacePreservation(i:WorkspacePreservationInput<'prepareWorkspacePreservation'>){return transaction(db,()=>{const r=owner(required(i.operationId),i.permit);if(r.request_hash!==i.requestHash)fail('WORKSPACE_PRESERVATION_REQUEST_CHANGED');validateManifest(r,i.manifest);const serialized=canonical(i.manifest);if(r.manifest_json){if(r.manifest_json!==serialized)fail('WORKSPACE_PRESERVATION_MANIFEST_CHANGED');return view(r);}current(r,i.permit);if(r.state!=='SENT')fail('WORKSPACE_PRESERVATION_CLOSED');
   const materials=[{key:'manifest',sha256:digest(serialized),byteSize:Buffer.byteLength(serialized)},...i.manifest.snapshot.entries.filter(e=>e.kind==='FILE').map(e=>({key:workspacePreservationMaterialKey(e.path),sha256:e.sha256,byteSize:e.byteSize}))];
   exec("UPDATE workspace_preservations SET state='PREPARED',manifest_json=?,manifest_hash=? WHERE operation_id=?",serialized,digest(serialized),i.operationId);for(const item of materials)exec('INSERT INTO workspace_preservation_materials(operation_id,material_key,expected_hash,byte_size,write_key) VALUES(?,?,?,?,?)',i.operationId,item.key,item.sha256,item.byteSize,randomUUID());return view(required(i.operationId));});},
  listWorkspacePreservationMaterials(i:WorkspacePreservationInput<'listWorkspacePreservationMaterials'>){owner(required(i.operationId),i.permit);return all('SELECT * FROM workspace_preservation_materials WHERE operation_id=? AND material_key>? ORDER BY material_key LIMIT ?',i.operationId,i.afterKey,i.limit).map(materialView);},
  completeWorkspacePreservation(i:WorkspacePreservationInput<'completeWorkspacePreservation'>){return transaction(db,()=>{const r=owner(required(i.operationId),i.permit),record=view(r),m=record.manifest,ack=i.acknowledgement;if(!m||r.request_hash!==i.requestHash||ack.manifestSha256!==r.manifest_hash||ack.snapshotSha256!==m.snapshot.sha256||!same(ack.repositoryIdentity,m.repositoryIdentity)||!same(ack.workspaceIdentity,m.workspaceIdentity)||!same(ack.parentIdentity,m.snapshot.identity))fail('WORKSPACE_PRESERVATION_ACK_UNPROVEN');if(r.state==='ACKNOWLEDGED'){if(!same(record.acknowledgement,ack))fail('WORKSPACE_PRESERVATION_ACK_IMMUTABLE');return record;}if(r.state!=='PREPARED')fail('WORKSPACE_PRESERVATION_CLOSED');current(r,i.permit);if(contentReady(i.operationId)!==ack.materialCount)fail('WORKSPACE_PRESERVATION_ACK_UNPROVEN');exec("UPDATE workspace_preservations SET state='ACKNOWLEDGED',acknowledgement_json=?,acknowledged_at=? WHERE operation_id=?",canonical(ack),new Date().toISOString(),i.operationId);return view(required(i.operationId));});},
  assertWorkspacePreservationReclaimable(i:WorkspacePreservationInput<'assertWorkspacePreservationReclaimable'>){return transaction(db,()=>{const r=owner(required(i.operationId),i.permit),record=view(r);if(r.state!=='ACKNOWLEDGED'||r.manifest_hash!==i.manifestSha256)fail('WORKSPACE_PRESERVATION_ACK_REQUIRED');current(r,i.permit);if(record.basis.fileOperations.some(f=>!f.cleanupConfirmed||!['SUCCEEDED','FAILED'].includes(f.state)))fail('WORKSPACE_PRESERVATION_FILE_UNKNOWN');if(contentReady(i.operationId)!==record.acknowledgement!.materialCount)fail('WORKSPACE_PRESERVATION_ACK_UNPROVEN');return record;});},
 };
 return {
  authorize(principalId:string,t:Target){const r=required(t.operationId);membership(String(r.project_id),principalId,true);if(r.principal_id!==principalId)fail('WORKSPACE_PRESERVATION_NOT_FOUND',404);return material(t);},
  validateWrite(i:{meta:{principalId:string;scopeId:string;idempotencyKey:string};target:Target;expectedHash:string;expectedSize:number}){const r=this.authorize(i.meta.principalId,i.target);if(i.meta.scopeId!==i.target.operationId||i.meta.idempotencyKey!==r.write_key||i.expectedHash!==r.expected_hash||i.expectedSize!==r.byte_size)fail('WORKSPACE_PRESERVATION_WRITE_CHANGED');},
  commit(principalId:string,t:Target,hash:string,writeId:string){const r=this.authorize(principalId,t);if(r.expected_hash!==hash||r.content_hash&&r.write_id!==writeId)fail('WORKSPACE_PRESERVATION_WRITE_CHANGED');exec('UPDATE workspace_preservation_materials SET content_hash=?,write_id=? WHERE operation_id=? AND material_key=?',hash,writeId,t.operationId,t.materialKey);return {operationId:t.operationId,materialKey:t.materialKey,contentHash:hash};},
  resolve(principalId:string,t:Target){const r=required(t.operationId);membership(String(r.project_id),principalId,false);const m=material(t);if(!m.content_hash)fail('WORKSPACE_PRESERVATION_MATERIAL_UNAVAILABLE');return {content_hash:m.content_hash};},
  execute(method:WorkspacePreservationMethod,input:unknown):unknown{return (handlers[method] as (i:unknown)=>unknown)(parseContract(workspacePreservationRequests[method],input));},
 };
}
