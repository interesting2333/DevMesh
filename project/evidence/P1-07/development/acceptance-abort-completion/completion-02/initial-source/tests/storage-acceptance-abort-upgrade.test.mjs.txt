test('v39 live proof survives v40 abort migration; DDL and history failures restore exact prior state',async t=>{
 // Historical upstream/Runner facts are explicit fixtures. The actual SQLite
 // migration must retain every old row and must not synthesize an abort.
 const f=await previewAccessFixture(t,{seedOptions:{targetVersion:39}}),at=new Date().toISOString(),operationId='op_v39_abort',policy={schemaVersion:'devmesh.acceptance-backend.v1',backendPrincipalId:'acceptance.v39.service',backendRef:'acceptance.v39.backend',runnerId:'runner.v39',templateVersion:'web-crud-v1',schemaRevision:'web-crud-v1.schema.1',initialModes:['EMPTY'],allowPreserveMigration:false,allowReplacement:false},registered=await f.call('registerAcceptanceBackendPolicy',{principalId:'user_1',projectId:f.p.projectId,policy}),image=(await f.readEffect(f.p.operationId)).request.imageIdentity;
 const dataPlan={schemaVersion:'devmesh.data-plan.v1',runId:f.p.runId,codeCommit:image.targetCommit,requirementRevision:1,planRevision:1,templateVersion:policy.templateVersion,checkConfigHash:hash('explicit historical config'),strategy:'INITIAL',initialData:'EMPTY',sourceReleaseId:null,migrationRef:null,replacementDecisionId:null},request=acceptancePreparationRequest.parse({schemaVersion:'devmesh.acceptance-prepare.v1',operationId,releaseId:'release_v39_abort',projectId:f.p.projectId,runId:f.p.runId,candidateId:f.candidate.candidateId,candidateRevision:1,manifestHash:f.candidate.manifestHash,reviewHash:hash('explicit historical review'),expectedAcceptedReleaseId:null,dataPlanHash:hash(canonical(dataPlan)),runEpoch:f.get('SELECT run_epoch FROM runs').run_epoch,requirementRevision:1,planRevision:1,codeCommit:image.targetCommit,image,dataPlan,policyHash:registered.policyHash,backendPrincipalId:policy.backendPrincipalId,backendRef:policy.backendRef,runnerId:policy.runnerId,source:null,target:{datasetId:'dataset_v39_abort',storageKey:f.p.projectId+'/dataset_v39_abort/app.sqlite',schemaRevision:policy.schemaRevision}}),requestHash=hash(canonical(request));
 f.run('INSERT INTO operations VALUES(?,?,?,?,?,?,?,?,?,?)',operationId,'user_1',f.p.runId,randomUUID(),hash(operationId),'explicit-v39-fixture','SUCCEEDED',null,at,at);f.run("INSERT INTO acceptance_preparations VALUES(?,?,?,?,?,?,?,?,?,?,'SENT',2,NULL,?,?)",operationId,'user_1',f.p.projectId,f.p.runId,f.candidate.candidateId,registered.policyHash,request.releaseId,request.target.datasetId,requestHash,canonical(request),at,at);f.run('INSERT INTO acceptance_preparation_history VALUES(?,?,?,?)',operationId,2,canonical({state:'SENT',revision:2,errorCode:null,observedAt:at}),at);

  const selected=failure==='ddl'?[...migrations.slice(0,39),{...migrations[39],sql:migrations[39].sql+'\nSELECT * FROM explicit_abort_upgrade_cut;'}]:migrations;

 }

 const next='op_v40_still_unresolved';f.run('INSERT INTO operations VALUES(?,?,?,?,?,?,?,?,?,?)',next,'user_1',f.p.runId,randomUUID(),hash(next),'explicit-v40-fixture','SUCCEEDED',null,at,at);assert.throws(()=>f.run("INSERT INTO acceptance_preparations SELECT ?,principal_id,project_id,run_id,candidate_id,policy_hash,'release_v40_blocked','dataset_v40_blocked',request_hash,request_json,'PREPARED',1,NULL,created_at,updated_at FROM acceptance_preparations WHERE operation_id=?",next,operationId),/ACCEPTANCE_PREPARATION_UNRESOLVED/);
